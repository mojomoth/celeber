+function padToSquare(canvas: HTMLCanvasElement, data: ImageData): { canvas: HTMLCanvasElement; offX: number; offY: number; side: number } | null {
+  const W = canvas.width;
+  const H = canvas.height;
+  const side = Math.max(W, H);
+  if (side === Math.min(W, H)) return null;
+  let r = 0;
+  let g = 0;
+  let b = 0;
+  let n = 0;
+  const add = (x: number, y: number) => {
+    const i = (y * W + x) * 4;
+    r += data.data[i];
+    g += data.data[i + 1];
+    b += data.data[i + 2];
+    n++;
+  };
+  for (let y = 0; y < H; y += 4) {
+    add(0, y);
+    add(W - 1, y);
+  }
+  for (let x = 0; x < W; x += 4) {
+    add(x, 0);
+    add(x, H - 1);
+  }
+  const c = document.createElement('canvas');
+  c.width = side;
+  c.height = side;
+  const ctx = c.getContext('2d');
+  if (!ctx) return null;
+  ctx.fillStyle = `rgb(${Math.round(r / n)},${Math.round(g / n)},${Math.round(b / n)})`;
+  ctx.fillRect(0, 0, side, side);
+  const offX = Math.round((side - W) / 2);
+  const offY = Math.round((side - H) / 2);
+  ctx.drawImage(canvas, offX, offY);
+  return { canvas: c, offX, offY, side };

    91	export function personBox(kp: readonly PosePoint[]): MBox {
    92	  const pts = kp.filter((p) => p.v >= VIS);

    95	  let x1 = Math.max(...pts.map((p) => p.x));

   100	  x1 += pad;

   105	export function clampBox(b: MBox): MBox {
   106	  const x = Math.max(0, Math.min(1, b.x));

   109	  const y1 = Math.max(0, Math.min(1, b.y + b.h));
   110	  return { x, y, w: Math.max(0, x1 - x), h: Math.max(0, y1 - y) };

   123	 * halfWidth는 이미지 너비 기준 정규화 값.

   125	export function thickSegment(a: P2, b: P2, halfWidth: number, aspect: number): P2[] {
   126	  // 픽셀 비례 공간으로 변환 (x * aspect)
   127	  const ax = a.x * aspect;
   128	  const bx = b.x * aspect;
   129	  const dx = bx - ax;
   130	  const dy = b.y - a.y;
   131	  const len = Math.hypot(dx, dy) || 1e-6;
   132	  const nx = (-dy / len) * halfWidth * aspect;
   133	  const ny = (dx / len) * halfWidth * aspect;
   134	  const toN = (x: number, y: number): P2 => ({ x: x / aspect, y });
   135	  return [toN(ax + nx, a.y + ny), toN(bx + nx, b.y + ny), toN(bx - nx, b.y - ny), toN(ax - nx, a.y - ny)];
   136	}

   151	export function outfitRegions(kp: readonly PosePoint[], aspect: number): OutfitRegionPolys[] {
   152	  const out: OutfitRegionPolys[] = [];
   153	  const ls = kp[POSE.leftShoulder];
   154	  const rs = kp[POSE.rightShoulder];
   155	  const lh = kp[POSE.leftHip];
   156	  const rh = kp[POSE.rightHip];
   157	  if (!(inFrame(ls) && inFrame(rs))) return out;
   158

   161	  const hipsVisible = inFrame(lh) && inFrame(rh);
   162	  const down = { x: 0, y: Math.min(1, shoulderW * aspect * 1.4) };
   163	  const lhp: P2 = hipsVisible ? lh : { x: ls.x, y: Math.min(1, ls.y + down.y) };
   164	  const rhp: P2 = hipsVisible ? rh : { x: rs.x, y: Math.min(1, rs.y + down.y) };
   165

   167	  const tl = lerp(ls, lhp, 0.12);
   168	  const tr = lerp(rs, rhp, 0.12);
   169	  const bl = lerp(ls, lhp, 0.95);
   170	  const br = lerp(rs, rhp, 0.95);
   171	
   172	  const center = [lerp(tl, tr, 0.33), lerp(tl, tr, 0.67), lerp(bl, br, 0.67), lerp(bl, br, 0.33)];
   173	  out.push({ key: 'torsoCenter', polys: [center] });
   174	  const sideL = [lerp(tl, tr, 0.02), lerp(tl, tr, 0.22), lerp(bl, br, 0.22), lerp(bl, br, 0.02)];
   175	  const sideR = [lerp(tl, tr, 0.78), lerp(tl, tr, 0.98), lerp(bl, br, 0.98), lerp(bl, br, 0.78)];
   176	  out.push({ key: 'torsoSides', polys: [sideL, sideR] });
   177	
   178	  const armHalf = shoulderW * 0.12;
   179	  const arms: P2[][] = [];
   180	  for (const [s, e] of [
   181	    [POSE.leftShoulder, POSE.leftElbow],
   182	    [POSE.rightShoulder, POSE.rightElbow],
   183	  ] as const) {
   184	    if (inFrame(kp[s]) && inFrame(kp[e])) arms.push(thickSegment(lerp(kp[s], kp[e], 0.15), lerp(kp[s], kp[e], 0.85), armHalf, aspect));
   185	  }
   186	  if (arms.length) out.push({ key: 'arms', polys: arms });
   187	
   188	  if (hipsVisible) {

   191	    const legs: P2[][] = [];
   192	    for (const [h, k, a] of [
   193	      [POSE.leftHip, POSE.leftKnee, POSE.leftAnkle],
   194	      [POSE.rightHip, POSE.rightKnee, POSE.rightAnkle],
   195	    ] as const) {

   199	    }
   200	    if (legs.length) out.push({ key: 'legs', polys: legs });

   211	    const ys = pts.map((p) => p.y);
   212	    const pad = shoulderW * 0.08;
   213	    const x0 = Math.min(...xs) - pad;
   214	    const x1 = Math.max(...xs) + pad;
   215	    const y0 = Math.min(...ys) - pad * aspect * 0.3;
   216	    const y1 = Math.max(...ys) + pad * aspect * 0.6;
   217	    feet.push([
   218	      { x: x0, y: y0 },
   219	      { x: x1, y: y0 },
   220	      { x: x1, y: y1 },
   221	      { x: x0, y: y1 },
   222	    ]);
   223	  }
   224	  if (feet.length) out.push({ key: 'feet', polys: feet });

   228	/**
   229	 * 체형 비율 추정 (전신 사진에서만 의미 있음). 정면 직립 가정의 근사치다.
   230	 * - 머리 높이: 얼굴 상자 높이 × 1.35 (이마 위 두피 포함), 얼굴이 없으면 코→어깨 중점 수직 거리 × 1.15
   231	 * - 키: 머리 꼭대기 → 발뒤꿈치(또는 발목) 최저점
   232	 * - 상하체 비: (골반 - 머리 꼭대기) / (발 - 골반)
   233	 * - 어깨/골반 폭 비: 어깨 관절 폭 / 골반 관절 폭 (관절 기준이라 실제 외곽선 비보다 크게 나온다)
   234	 */
   235	export function estimateBodyRatios(kp: readonly PosePoint[], aspect: number, faceBox?: MBox): BodyMeasurement | undefined {
   236	  const vis = poseVisibility(kp);
   237	  if (!(vis.shoulders && vis.hips && vis.ankles)) return undefined;
   238	  const ls = kp[POSE.leftShoulder];
   239	  const rs = kp[POSE.rightShoulder];
   240	  const lh = kp[POSE.leftHip];
   241	  const rh = kp[POSE.rightHip];
   242	  const shoulderMid = mid(ls, rs);
   243	  const hipMid = mid(lh, rh);
   244	  const footYs = [POSE.leftHeel, POSE.rightHeel, POSE.leftAnkle, POSE.rightAnkle, POSE.leftFoot, POSE.rightFoot]
   245	    .map((i) => kp[i])
   246	    .filter((p) => inFrame(p))
   247	    .map((p) => p.y);
   248	  if (!footYs.length) return undefined;
   249	  const footY = Math.max(...footYs);
   250	
   251	  let headH: number;
   252	  let headTop: number;
   253	  if (faceBox && faceBox.h > 0) {
   254	    // FACE_OVAL 윗점(10번)은 이마 위쪽이라 정수리까지 약 35%를 더한다
   255	    headH = faceBox.h * 1.35;
   256	    headTop = faceBox.y - faceBox.h * 0.35;
   257	  } else {
   258	    const nose = kp[POSE.nose];
   259	    if (!inFrame(nose)) return undefined;
   260	    // 정면 직립에서 코→어깨선 수직 거리 ≈ 머리 높이의 0.87배
   261	    headH = Math.abs(shoulderMid.y - nose.y) * 1.15;
   262	    headTop = nose.y - headH * 0.55;
   263	  }
   264	  const height = footY - headTop;
   265	  if (height <= 0 || headH <= 0) return undefined;
   266	  const shoulderW = Math.hypot((ls.x - rs.x) * aspect, ls.y - rs.y);
   267	  const hipW = Math.hypot((lh.x - rh.x) * aspect, lh.y - rh.y);
   268	  const r2 = (v: number) => Math.round(v * 100) / 100;
   269	  return {
   270	    headsTall: r2(height / headH),
   271	    upperToLower: r2((hipMid.y - headTop) / Math.max(1e-6, footY - hipMid.y)),
   272	    legToHeight: r2((footY - hipMid.y) / height),
   273	    shoulderToHip: hipW > 0 ? r2(shoulderW / hipW) : undefined,
   274	  };
   275	}

   287	 * 크기: 어깨 폭과 코→어깨 거리 중 큰 쪽 기준, 원본 픽셀 비율을 유지한다.

   289	export function headCrop(kp: readonly PosePoint[], width: number, height: number): MBox | null {
   290	  const nose = kp[POSE.nose];

   292	  const rs = kp[POSE.rightShoulder];

   297	  if (inFrame(ls) && inFrame(rs)) {

   304	  }

   307	  const y0 = n.y - size * 0.55;
   308	  return clampBox({ x: x0 / width, y: y0 / height, w: size / width, h: size / height });

    54	/** 사람 몸에서 나올 수 있는 비율 범위 (camera-body RATIO_BOUNDS와 같은 값). 벗어나면 전신 비율은 측정하지 않은 것으로 둔다. */
    55	const BODY_BOUNDS = { headsTall: [3, 13], upperToLower: [0.3, 3], legToHeight: [0.2, 0.85], shoulderToHip: [0.4, 3] } as const;
    56	export function plausibleBody(b: NonNullable<VibeMeasurement['body']>): boolean {
    57	  return (Object.keys(BODY_BOUNDS) as (keyof typeof BODY_BOUNDS)[]).every((k) => {
    58	    const v = b[k];
    59	    return v === undefined || (Number.isFinite(v) && v >= BODY_BOUNDS[k][0] && v <= BODY_BOUNDS[k][1]);
    60	  });

    63	/** 측정값/통계 → 규칙 초안. force=true면 '여러 명' 등을 경고로 낮춰 진행 */
    64	export function buildRuleDraft(input: VibeMeasurement | undefined, stats: ServerImageStats | undefined, opts: { force?: boolean } = {}): RuleDraft {
    65	  const resolved = resolveMeasurement(input, opts);
    66	  const m = resolved.measurement;
    67	  let analyzability = judgeAnalyzability(m, stats, { fallback: resolved.fallback });
    68	  if (opts.force) analyzability = forceAnalyzable(analyzability);
    69	  // 브라우저 측정이 있으면 측정 영역만 쓴다 (상반신/얼굴 사진에서 중앙 하단을 하의로 오인하지 않도록)
    70	  const outfit = m ? ruleOutfitFromMeasurement(m) : stats ? ruleOutfitFromStats(stats) : { items: [], clusters: [], notes: [] };
    71	  if (!analyzability.partial.outfit) outfit.items = outfit.items.filter((i) => i.category !== 'shoes');
    72	  const palette = rulePalette(outfit.clusters, m?.face?.lips?.rgb);
    73	  const styleKeywords = ruleStyleKeywords(palette);
    74	  const frameLabel = frameLabelFromRatio(m?.body?.shoulderToHip);
    75	  const body: VibeAnalysis['body'] =
    76	    m?.body && m.pose?.framing === 'full-body' && plausibleBody(m.body)
    77	      ? {
    78	          measured: true,
    79	          headsTall: m.body.headsTall,
    80	          upperToLower: m.body.upperToLower,
    81	          legToHeight: m.body.legToHeight,
    82	          shoulderToHip: m.body.shoulderToHip,
    83	          frameLabel,
    84	          notes: ['PoseLandmarker 관절 좌표 기반 근사치입니다. 원근·자세·신발 굽에 따라 달라집니다.', '어깨/골반 비는 관절 사이 거리라 실제 외곽선 비보다 크게 나옵니다.'],
    85	        }
    86	      : undefined;
    87	  const silhouette = ruleSilhouette(outfit.items, analyzability.framing, body?.frameLabel);
    88	  return {
    89	    measured: !!m?.face,
    90	    measurement: m,
    91	    measurementDropped: !!input && !m,
    92	    forced: !!opts.force,
    93	    analyzability,
    94	    outfit,
    95	    palette,
    96	    styleKeywords,
    97	    silhouette,
    98	    summary: ruleOutfitSummary(outfit.items, styleKeywords),
    99	    mood: ruleMood(styleKeywords, palette),
   100	    lookName: ruleLookName(styleKeywords, palette),
   101	    personalColor: rulePersonalColor(m, stats, analyzability.partial.personalColor),
   102	    body,
   103	  };
   104	}

   189	const toNamed = (c: { hex: string; name: string }, fallbackHex?: string): NamedColor | null => {
   190	  const hex = normalizeHex(c.hex) ?? (fallbackHex ? normalizeHex(fallbackHex) : null);
   191	  if (!hex) return null;
   192	  return { hex, name: c.name?.trim() || colorNameKo(hex) };

   195	function sanitizeBox(b: { x: number; y: number; w: number; h: number } | undefined) {
   196	  if (!b) return undefined;
   197	  const x = clamp01(b.x);
   198	  const y = clamp01(b.y);
   199	  const w = Math.min(1 - x, clamp01(b.w));
   200	  const h = Math.min(1 - y, clamp01(b.h));
   201	  if (w <= 0.005 || h <= 0.005) return undefined;
   202	  const r = (v: number) => Math.round(v * 1000) / 1000;
   203	  return { x: r(x), y: r(y), w: r(w), h: r(h) };
   204	}

   206	/**
   207	 * AI 아이템 정리: hex 검증, box 정규화, 규칙 초안 box/색으로 보강.
   208	 * @param opts.measured 규칙 초안 색이 브라우저 측정 영역에서 나온 값인지. 아니면(사진 중앙 통계·인물 없음) 사진에 없는 색일 수 있어 비교 노트를 달지 않는다
   209	 */
   210	export function sanitizeAiItems(ai: VibeAi['outfit']['items'], rule: OutfitItem[], opts: { measured?: boolean } = {}): OutfitItem[] {
   211	  const measured = opts.measured ?? true;
   212	  const out: OutfitItem[] = [];
   213	  for (const it of ai) {
   214	    const ruleSame = rule.find((r) => r.category === it.category);
   215	    const color = toNamed(it.color, ruleSame?.color.hex);
   216	    if (!color) continue;
   217	    const secondaryColors = it.secondaryColors.map((c) => toNamed(c)).filter((c): c is NamedColor => !!c);
   218	    const box = sanitizeBox(it.box) ?? ruleSame?.box;
   219	    const notes: string[] = [];
   220	    if (it.note.trim()) notes.push(it.note.trim());
   221	    // AI 노트가 이미 측정색·조명 차이를 말했으면 같은 내용을 한 번 더 붙이지 않는다
   222	    const aiMentionsGap = /측정|조명/.test(it.note);
   223	    if (measured && !aiMentionsGap && ruleSame && deltaE2000(hexToLab(ruleSame.color.hex), hexToLab(color.hex)) > 12) notes.push(`측정색 ${ruleSame.color.name}(${ruleSame.color.hex})과 차이가 있습니다(조명 영향 가능).`);
   224	    out.push({
   225	      category: it.category,
   226	      name: it.name.trim() || ruleSame?.name || it.category,
   227	      color,
   228	      secondaryColors: secondaryColors.length ? secondaryColors : undefined,
   229	      material: it.material.trim() || undefined,
   230	      fit: it.fit.trim() || undefined,
   231	      pattern: it.pattern.trim() || undefined,
   232	      box,
   233	      note: notes.length ? notes.join(' ') : undefined,
   234	    });
   235	  }
   236	  return out;
   237	}

   246	/** AI 퍼스널컬러 + 규칙 측정 병합 */
   247	export function mergePersonalColor(rule: RulePersonalColor, ai: VibeAi['personalColor'] | null, conditionsOk: boolean, grayscale: boolean): VibePersonalColor {
   248	  const r = rule.pc;
   249	  if (!ai) return r;
   250	  const aiTone = ai.tone !== 'unknown' ? ai.tone : undefined;
   251	  const aiEvidence = ai.evidence.map((e) => e.trim()).filter(Boolean);
   252	  const caveat = [r.lightingCaveat, ai.lightingCaveat.trim()].filter(Boolean).join(' ');
   253	  if (grayscale) return { ...r, lightingCaveat: caveat };
   254	
   255	  if (r.measured && r.tone) {
   256	    // 측정 기반 분류가 기준. AI는 교차 검증
   257	    const agreement = toneAgreement(r.tone, aiTone);
   258	    const cls = rule.classification;
   259	    const conf = vibeColorConfidence({
   260	      measured: true,
   261	      conditionsOk,
   262	      skinUsed: undefined,
   263	      classifierMargin: cls?.margin,
   264	      agreement,
   265	      hasHair: !!r.hair,
   266	      hasEyes: !!r.eyes,
   267	    });
   268	    // 측정 기반 신뢰도(초안)에 일치도 보정만 반영
   269	    const delta = agreement === true ? 0.1 : agreement === 'season-only' ? 0.04 : agreement === false ? -0.1 : 0;
   270	    const score = Math.max(0.05, Math.min(conditionsOk ? 0.85 : 0.3, Math.round((r.confidence + delta) * 100) / 100));
   271	    const aiLine = aiTone ? `AI 판단: ${getToneProfile(aiTone).nameKo}${agreement === true ? ' (측정 판정과 일치)' : agreement === 'season-only' ? ' (계절 일치)' : ' (측정 판정과 다름)'}` : 'AI는 톤을 판단하지 않았습니다.';
   272	    const agreementFactor = conf.factors.find((f) => f.includes('AI 판단'));
   273	    return {
   274	      ...r,
   275	      confidence: score,
   276	      evidence: [...r.evidence, aiLine, ...(agreementFactor ? [agreementFactor] : []), ...aiEvidence.slice(0, 3)],
   277	      lightingCaveat: caveat,
   278	    };
   279	  }
   280	
   281	  // 측정 없음 → AI 판단 기준. 축은 AI가 본 피부/헤어/눈 색으로 분류기를 돌려 계산 (갭 비교용)
   282	  // 규칙 초안의 색·톤은 사진 중앙 픽셀(옷·배경일 수 있음)이라, AI가 비운 값을 그것으로 되살리지 않는다
   283	  const skinHex = normalizeHex(ai.skinHex);
   284	  const hairHex = normalizeHex(ai.hairHex);
   285	  const eyeHex = normalizeHex(ai.eyeHex);
   286	  const skin = skinHex ? describeColor(skinHex, 'skin') : undefined;
   287	  const hair = hairHex ? describeColor(hairHex) : undefined;
   288	  const eyes = eyeHex ? describeColor(eyeHex) : undefined;
   289	  const tone = aiTone;
   290	  if (!tone && !skin) return notJudgedPersonalColor(aiEvidence, caveat, 'AI가 사진에서 피부색과 톤을 판단하지 않았습니다.');
   291	  const cls = skin ? classifyMeasuredColors({ skin: skin.lab, hair: hair?.lab, eyes: eyes?.lab }) : undefined;
   292	  const conf = vibeColorConfidence({ measured: false, conditionsOk, agreement: toneAgreement(cls?.tone, aiTone), hasHair: !!hair, hasEyes: !!eyes, aiConfidence: clamp01(ai.confidence) });
   293	  return reconcileUndertone({
   294	    measured: false,
   295	    skin,
   296	    hair,
   297	    eyes,
   298	    season: tone ? toneSeason(tone) : undefined,
   299	    tone,
   300	    toneNameKo: tone ? getToneProfile(tone).nameKo : undefined,
   301	    undertone: ai.undertone !== 'unknown' ? ai.undertone : tone ? cls?.undertone : undefined,
   302	    axes: cls?.axes,
   303	    confidence: tone ? conf.score : 0,
   304	    evidence: [...aiEvidence, ...conf.factors, ...(skinHex ? ['피부·헤어·눈 색은 AI가 사진에서 읽은 값이며 픽셀 측정값이 아닙니다.'] : [])],
   305	    lightingCaveat: caveat,
   306	  });
   307	}

   310	export function notJudgedPersonalColor(evidence: string[], lightingCaveat: string, reason: string): VibePersonalColor {
   311	  return { measured: false, confidence: 0, evidence: [reason, ...evidence.filter((e) => e !== reason)], lightingCaveat };

   292	  const rs = kp[POSE.rightShoulder];

   299	    const b = px(rs);

   304	  }
   305	  if (size < 16) return null;
   306	  const x0 = n.x - size / 2;
   307	  const y0 = n.y - size * 0.55;
   308	  return clampBox({ x: x0 / width, y: y0 / height, w: size / width, h: size / height });
