    43	  const bad = nearestInPalette(hex, [...p.worst, ...p.drapeWorst]).deltaE;
    44	  return Math.round((1 / (1 + Math.exp(-(bad - good) / 6))) * 100) / 100;
    45	}

    86	/** 색이 강한 인공 조명 (화이트밸런스 보정이 크게 개입하는 광원) */
    87	const HARSH_LIGHT = new Set(['incandescent', 'warm-led', 'fluorescent', 'mixed']);
    88	/** 피부 부위 간 색 편차가 이 값(ΔE)을 넘으면 조명 얼룩/색 번짐으로 본다 */
    89	export const SKIN_SPREAD_UNRELIABLE_DE = 15;
    90	
    91	/**
    92	 * 측정 조건이 판정을 믿기 어려운 수준인지.
    93	 * 색이 강한 조명(백열·전구색 LED·형광·혼합, 또는 3,800K 미만/8,000K 초과, 조명 품질 0.45 미만)에서
    94	 * 피부 부위 간 편차까지 크면, 화이트밸런스 보정이 과하거나 부족해 웜/쿨이 뒤집힐 수 있다.
    95	 * 이때 분류기 1·2위 차가 커도(확신이 높아 보여도) 신뢰도가 '보통'으로 나오지 않게 한다.
    96	 */
    97	export function unreliableCapture(face: FaceAnalysis): { unreliable: boolean; reason: string } {
    98	  const env = face.environment;
    99	  const harsh = HARSH_LIGHT.has(env.lightSource) || env.cctK < 3800 || env.cctK > 8000 || env.lightingQuality < 0.45;
   100	  const spread = face.skin.regionSpreadDeltaE;
   101	  const unreliable = harsh && spread > SKIN_SPREAD_UNRELIABLE_DE;
   102	  return {
   103	    unreliable,
   104	    reason: unreliable
   105	      ? `${env.lightSourceLabel}(${env.cctK.toLocaleString('en-US')}K) 아래에서 이마·볼·턱 색 편차가 ΔE ${spread.toFixed(1)}로 커서 웜/쿨 판정이 흔들릴 수 있습니다. 자연광 정면에서 다시 촬영해 주세요.`
   106	      : '',
   107	  };
   108	}

   110	/**
   111	 * 조명 자체가 판정을 흔들 수 있는 촬영인지 (부위 편차와 무관하게).
   112	 * 환경 단계가 '재촬영 권장'을 냈거나, 조명 품질이 0.45 미만이거나, 색이 강한 인공 조명이면 true.
   113	 * 이때 ③ 웜/쿨 %와 ⑭ 타입 구분 명확도를 조명 신뢰도(trust)만큼 50:50 쪽으로 줄이고, 점수 상한을 둔다.
   114	 */
   115	export function lightingCaution(face: FaceAnalysis): { flagged: boolean; trust: number; reason: string } {
   116	  const env = face.environment;
   117	  const harsh = HARSH_LIGHT.has(env.lightSource) || env.cctK < 3800 || env.cctK > 8000;
   118	  const flagged = env.retakeRecommended || env.lightingQuality < 0.45 || harsh;
   119	  if (!flagged) return { flagged: false, trust: 1, reason: '' };
   120	  const lq = Math.max(0, Math.min(1, env.lightingQuality));
   121	  const trust = Math.round((0.4 + 0.4 * Math.min(1, lq / 0.75)) * 100) / 100;
   122	  const why = [
   123	    env.retakeRecommended ? '환경 촬영에서 재촬영 권장' : '',
   124	    `${env.lightSourceLabel} ${env.cctK.toLocaleString('en-US')}K`,
   125	    `조명 품질 ${Math.round(lq * 100)}점`,
   126	  ]
   127	    .filter(Boolean)
   128	    .join(', ');
   129	  return {
   130	    flagged,
   131	    trust,
   132	    reason: `조명 확인 필요: ${why}. ${harsh ? '색이 있는 조명에서는' : '노출·조명 상태가 나쁘면'} 보정 후에도 웜/쿨이 바뀔 수 있어 웜/쿨 비율과 타입 확신도를 낮춰 표시했습니다. 자연광 정면에서 다시 촬영해 보세요.`,
   133	  };
   134	}

   136	/** 조명 신뢰도만큼 웜 %를 50 쪽으로 당긴다 (합 100 유지) */
   137	export function softenWarmPercent(warm: number, trust: number): { warm: number; cool: number } {
   138	  const w = Math.round(50 + (warm - 50) * Math.max(0, Math.min(1, trust)));
   139	  return { warm: w, cool: 100 - w };
   140	}

   145	/** ⑭ 실제 측정 품질에서 신뢰도 계산 (고정값 없음) */
   146	export function reportConfidence(face: FaceAnalysis, cls: PersonalColorClassification, aiTone: Tone | null | undefined): ReportConfidence {
   147	  const q = face.front.quality;
   148	  const detection = q.detectionScore * (q.passed ? 1 : 0.8);
   149	  const light = lightingCaution(face);
   150	  const base = computeReportConfidence({
   151	    lightingQuality: face.environment.lightingQuality,
   152	    whiteBalanceDeltaE: face.skin.whiteBalance.deltaE,
   153	    whiteBalanceMethod: face.skin.whiteBalance.method,
   154	    skinConfidence: face.skin.confidence,
   155	    faceDetectionScore: detection,
   156	    skinSpreadDeltaE: face.skin.regionSpreadDeltaE,
   157	    // 조명이 의심스러우면 1·2위 차를 그대로 믿지 않는다
   158	    classifierMargin: cls.margin * light.trust,
   159	    ruleAiAgreement: ruleAiAgreement(cls.tone, aiTone),
   160	    hasHair: !!face.featureColors.hair,
   161	    hasEyes: !!face.featureColors.eyes,
   162	  });
   163	  // 근거 문구에 실제 측정값을 덧붙인다
   164	  const env = face.environment;
   165	  const u = unreliableCapture(face);
   166	  const factors: ConfidenceFactor[] = base.factors.map((f) => {
   167	    if (f.key === 'lighting') return { ...f, detail: `${f.detail} (${env.lightSourceLabel} · ${env.cctK.toLocaleString('en-US')}K · 밝기 ${env.brightness.label})` };
   168	    if (f.key === 'face-detection') return { ...f, detail: `${f.detail} (검출 ${Math.round(q.detectionScore * 100)}%, 좌우 ${Math.round(face.front.yawDeg)}°)` };
   169	    if (f.key === 'classifier-margin') {
   170	      const lightNote = u.unreliable || light.flagged ? ' — 조명 영향으로 확신도를 낮춰 반영했습니다' : '';
   171	      const detail = light.flagged ? (f.score >= 0.57 ? '1순위 타입이 비교적 뚜렷합니다.' : '보조 타입과 점수가 가깝습니다.') : f.detail;
   172	      return { ...f, detail: `${detail} (1·2위 차 ${Math.round(cls.margin * 100)}%p)${lightNote}` };
   173	    }
   174	    if (f.key === 'skin-uniformity' && u.unreliable) return { ...f, detail: `${f.detail} (조명 얼룩 의심)` };
   175	    if (f.key === 'skin-measurement' && face.skin.confidenceReasons.length) return { ...f, detail: `${f.detail} (${face.skin.confidenceReasons.slice(0, 2).join(', ')})` };
   176	    return f;
   177	  });
   178	  if (u.unreliable) {
   179	    // 앱 단 감점: 점수의 70%, 최대 49 → 항상 '낮은 신뢰도'. 점수는 여전히 측정값에 따라 달라진다.
   180	    const score = Math.min(49, Math.round(base.score * 0.7));
   181	    return { score, grade: 'low', gradeLabel: '낮은 신뢰도 · 재촬영 권장', factors };
   182	  }
   183	  if (light.flagged) {
   184	    // 상한 55로 부드럽게 눌러(50 초과분은 10%만 반영) 측정 품질에 따른 순서는 유지한다.
   185	    const score = base.score <= 50 ? base.score : Math.min(FLAGGED_LIGHTING_SCORE_CAP, Math.round(50 + (base.score - 50) * 0.1));
   186	    const grade = score >= 50 ? 'medium' : 'low';
   187	    return { score, grade, gradeLabel: grade === 'medium' ? '보통 신뢰도 · 조명 확인 필요' : '낮은 신뢰도 · 조명 확인 필요', factors };
   188	  }
   189	  return { ...base, factors };
   190	}

    78	 * '이 타입에 어울리는 수준'(TONE_PROTOTYPES)을 먼저 쓰고, 측정값은 근거로 덧붙인다.
    79	 * 측정 축이 타입 권장과 30 이상 다르면 '측정은 …'을 함께 적어 숨기지 않는다.
    80	 */
    81	export function axisRowText(axis: 'value' | 'chroma' | 'clarity', cls: PersonalColorClassification, tone: Tone): string {
    82	  const idx = axis === 'value' ? 1 : axis === 'chroma' ? 2 : 4;
    83	  const rec = Math.round(TONE_PROTOTYPES[tone][idx] * 100);
    84	  const recLabel = bareLabel(axisLevelLabel(axis, rec));
    85	  const hint =
    86	    axis === 'value'
    87	      ? rec >= 60
    88	        ? '라이트 톤'
    89	        : rec >= 40
    90	          ? '미디엄 톤'
    91	          : '딥 톤'
    92	      : axis === 'chroma'
    93	        ? rec <= 35
    94	          ? '뮤트 톤이 잘 어울림'
    95	          : rec <= 65
    96	            ? '중간 채도가 무난함'
    97	            : '선명한 색이 잘 어울림'
    98	        : rec >= 60
    99	          ? '맑고 깨끗한 느낌'
   100	          : rec >= 40
   101	            ? '부드럽고 차분한 느낌'
   102	            : '그윽하고 차분한 느낌';
   103	  const m = cls.axes[axis];
   104	  const measuredNum =
   105	    axis === 'value' ? `측정 피부 L* ${Math.round(cls.inputs.skinLab.l)}` : axis === 'chroma' ? `측정 피부 C* ${Math.round(Math.hypot(cls.inputs.skinLab.a, cls.inputs.skinLab.b))}` : `측정 지수 ${m}`;
   106	  const diff = Math.abs(m - rec) >= 30 ? ` (측정은 ${bareLabel(axisLevelLabel(axis, m))})` : '';
   107	  return `${recLabel} (${hint}) · ${measuredNum}${diff}`;
   108	}

# personal-color-report — 내 얼굴과 전신(선택)에서 퍼스널컬러 추출

얼굴 분석(camera-face)과 전신 비율(camera-body, 선택)로 퍼스널컬러를 판정하고 리포트를 만든다. 리포트는 [레퍼런스 카드](../../docs/ui/personal-color-report-guide/navy-men-personal-color-report-card.jpeg)와 섹션 구조가 같다. 390pt 모바일에서는 세로 스크롤로 배치했다.

- 포트: 3105. API 네임스페이스는 `/api/report/*`다.
- 선행 PoC를 재노출한다. `/api/face/*`는 camera-face, `/api/body/*`는 camera-body의 Route Handler다.

## 실행

```bash
cd /Users/jyl/work/repo/celeber
pnpm --filter @celeber/personal-color-report dev      # http://localhost:3105
pnpm --filter @celeber/personal-color-report build && pnpm --filter @celeber/personal-color-report start
pnpm --filter @celeber/personal-color-report test     # vitest

```

## 화면 흐름

| 경로 | 내용 |
|---|---|
| `/` | 소개, `리포트 만들기`, 최근 리포트 5개 |
| `/new` | 입력 방법을 고른다: **새로 촬영**(환경→정면→측면→전신(선택)) 또는 **저장된 얼굴 분석 선택** + 전신 세션(선택 / 전신 없이 / 새 전신). 다음 화면 `진단 준비`에서 리포트 스타일(남성/여성/선택 안 함)을 고른다. 기본값은 사진 기반 `FaceAnalysis.ai.presentation`이다. 그다음 생성하면 `/reports/<id>`로 이동한다 |
| `/new?step=capture` | 바로 촬영. camera-face `FaceCaptureFlow` 다음에 camera-body `BodyCaptureFlow`(4/4 전신, 건너뛰기 가능)가 이어진다 |
| `/new?face=<id>&step=body` | 리포트의 `전신 사진 추가`에서 들어온다. 얼굴은 고정하고 전신부터 시작한다 |
| `/reports` | 리포트 목록. `리포트 전체 초기화`는 ConfirmDialog를 거친다. personal_color_reports만 지우고 얼굴·전신 분석과 사진은 남긴다 |
| `/reports/<id>` | 리포트 다시보기. 스타일 변형을 바꾸면 PATCH로 저장된다. `같은 얼굴로 다시 만들기`, 삭제(확인 단계) |

모든 카메라 단계에는 업로드 대체 경로가 있다. 선행 플로우가 제공한다.

## 알고리즘과 근거

1. **입력**
   - camera-face가 저장한 `FaceAnalysis`를 쓴다. 화이트밸런스 보정한 피부 Lab, ROI별 편차, 헤어/눈 Lab, 환경(광원·CCT·lightingQuality), 정면 랜드마크가 들어 있다.
   - 서버는 face_sessions와 body_sessions에서 다시 읽는다. 클라이언트가 보낸 측정값은 믿지 않는다.
2. **판정** — `classifyFace` → core `classifyPersonalColor`. 결정적이다.
   - 5축: 웜(색상각·b*), 명도(L*), 채도(C*), 대비(피부-헤어/눈 ΔL*), 청탁.
   - 12톤 원형과의 가중 거리를 softmax로 바꿔 1위를 판정, 2위를 보조 타입으로 삼는다.
   - **앱 단 보정**(core로 승격 후보):
     - camera-face의 헤어 L*는 그늘 픽셀 때문에 2~10으로 매우 낮게 나온다. core 기본값(ΔL* 18→0, 62→1)으로는 밝은 피부 + 검은 머리가 모두 대비 100이 된다.
     - 그 결과 L* 86 피부가 겨울 딥과 거의 동점이었다(margin 0.004).
     - 대비 범위를 ΔL* 35→0, 85→1로 넓히고 헤어/눈 L* 하한을 8로 두었다. 웜/쿨 판단은 그대로다. 같은 입력이 여름 뮤트(보조 여름 라이트, margin 0.13)가 된다.
     - 테스트로 고정했다.
3. **서술** — Claude(`callVisionStructured`, 구조화 출력, 얕은 zod 스키마 `AiNarrativeSchema`)에게 아래를 보낸다.
   - 입력: 정면 사진 + 측정값 + 규칙 판정.
   - 출력:
     - (1) 규칙과 독립적인 톤 의견(`independentTone`)
     - (2) 규칙 톤 기준 서술: 첫인상, 설명, 분석 포인트, 특징 값, TIP, 그루밍, 오늘 포인트
   - 빈 값은 템플릿으로 채운다(`mergeAiNarrative`).
   - 키가 없거나 실패하면 `templateNarrative`를 쓴다. 템플릿도 측정 색상각·b*·ITA·ΔL*를 문장에 넣는다.
4. **조립** — `assembleReport`. 카드 섹션에 1:1로 채우고 `PersonalColorReportSchema.parse`로 검증한 뒤 저장한다.
5. **드레이핑**
   - 크롭: `drapeCrop`가 같은 정면 사진에서 얼굴 위 22%부터 턱 아래 60%까지 잡는다. 사진이 턱 바로 아래에서 끝나면 35%까지만 잡는다.
   - 천 윤곽: `drapeOutline`이 어깨 → 목 옆 → 턱 아래 곡선을 만든다. 모든 점이 턱끝보다 아래라 얼굴을 가리지 않는다.
   - 그리기: `DrapeThumb`가 천 색 바탕, 목에 드리우는 그림자, 주름 음영을 그리고, 목에 비치는 반사광을 턱선 아래에만 soft-light(알파 0.12)로 더한다. 얼굴(턱선 위) 픽셀은 모든 색에서 같다 — 390px Chrome에서 베스트/워스트 캔버스의 위쪽 55% 픽셀 차이 0으로 확인.
   - 사진 밖 영역은 가장자리 픽셀로 채운다.
   - 색만 바꾸고 조명·구도는 원본 그대로다.
   - `harmony`는 추천 팔레트와 워스트 팔레트까지의 ΔE2000 차이를 시그모이드로 바꾼 값이다.
6. **신뢰도(⑭)** — `reportConfidence` → core `computeReportConfidence`. 고정값은 없다. 입력은 다음과 같다.
   - 환경 lightingQuality
   - 화이트밸런스 보정량 ΔE
   - 정면 검출 점수(품질 미통과 시 ×0.8)
   - 피부 ROI 편차 ΔE
   - 분류 margin
   - 규칙·AI 일치도(톤 일치 / 계절만 일치 / 불일치 / AI 없음이면 제외)
   - 헤어·눈 측정 여부

   요인별 근거에는 실제 값(광원·K·밝기, 검출 %, 1·2위 차)을 넣는다.

   앱 단 감점(`unreliableCapture`): 색이 강한 조명(백열·전구색 LED·형광·혼합, 3,800K 미만/8,000K 초과, 조명 품질 0.45 미만)에서 피부 부위 편차 ΔE가 15를 넘으면 점수 ×0.7(최대 49) → 항상 '낮은 신뢰도 · 재촬영 권장'. 13 주의사항 맨 앞과 ② 아래에 재촬영 안내를 넣는다. 분류 margin이 커서 확신이 높아 보여도 조명 때문에 웜/쿨이 뒤집혔을 수 있기 때문이다.
7. **전신** — camera-body `toBodyStyling`(체형·등신 → 실루엣 제안/회피)을 그대로 쓴다. 측정이 완료된 세션만 연결한다.
