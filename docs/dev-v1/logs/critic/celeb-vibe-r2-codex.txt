   304	    }
   305	    if (i.hasHair) s += 0.04;
   306	    if (i.hasEyes) s += 0.03;

   308	    s += Math.min(0.1, (i.classifierMargin ?? 0) * 0.5);
   309	  } else if (i.heuristic) {

   312	  } else {
   313	    s = 0.3 + 0.25 * Math.max(0, Math.min(1, i.aiConfidence ?? 0.5));
   314	    f.push('측정값 없이 AI가 사진을 보고 판단했습니다.');
   315	  }

   317	  const basis = i.measured ? '측정 기반 판정' : '색 규칙 추정';
   318	  if (i.agreement === true) {

   331	  }
   332	  s = Math.max(0.05, Math.min(CELEB_CONFIDENCE_CAP, s));
   333	  return { score: Math.round(s * 100) / 100, factors: f };

   194	function sanitizeBox(b: { x: number; y: number; w: number; h: number } | undefined) {
   195	  if (!b) return undefined;
   196	  const x = clamp01(b.x);
   197	  const y = clamp01(b.y);
   198	  const w = Math.min(1 - x, clamp01(b.w));
   199	  const h = Math.min(1 - y, clamp01(b.h));
   200	  if (w <= 0.005 || h <= 0.005) return undefined;
   201	  const r = (v: number) => Math.round(v * 1000) / 1000;
   202	  return { x: r(x), y: r(y), w: r(w), h: r(h) };
   203	}

    79	/** 정규화 사각형 영역의 픽셀 (필터 통과분) */
    80	export function rectPixels(img: Pixels, x0: number, y0: number, x1: number, y1: number, filter?: (r: number, g: number, b: number) => boolean, max = 6000): Rgb[] {
    81	  const { data, width, height } = img;
    82	  const px0 = Math.max(0, Math.floor(x0 * width));
    83	  const px1 = Math.min(width - 1, Math.ceil(x1 * width));
    84	  const py0 = Math.max(0, Math.floor(y0 * height));
    85	  const py1 = Math.min(height - 1, Math.ceil(y1 * height));
    86	  const area = Math.max(1, (px1 - px0 + 1) * (py1 - py0 + 1));
    87	  const step = Math.max(1, Math.floor(Math.sqrt(area / max)));
    88	  const out: Rgb[] = [];
    89	  for (let y = py0; y <= py1; y += step) {
    90	    for (let x = px0; x <= px1; x += step) {
    91	      const i = (y * width + x) * 4;
    92	      const a = data[i + 3] ?? 255;
    93	      if (a < ALPHA_MIN) continue;
    94	      const r = over(data[i], a);
    95	      const g = over(data[i + 1], a);
    96	      const b = over(data[i + 2], a);
    97	      if (filter && !filter(r, g, b)) continue;
    98	      out.push({ r, g, b });
    99	    }
   100	  }
   101	  return out;
   102	}

   105	  if (!px.length) return 0;
   106	  return px.reduce((s, p) => s + 0.299 * p.r + 0.587 * p.g + 0.114 * p.b, 0) / px.length;

    38	/** 매직 바이트로 이미지 MIME 판별 (확장자/헤더 신뢰 안 함) */
    39	export function sniffImageMime(buf: Buffer): string | null {
    40	  if (buf.length < 12) return null;
    41	  if (buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'image/jpeg';
    42	  if (buf[0] === 0x89 && buf[1] === 0x50 && buf[2] === 0x4e && buf[3] === 0x47) return 'image/png';
    43	  if (buf.toString('ascii', 0, 4) === 'RIFF' && buf.toString('ascii', 8, 12) === 'WEBP') return 'image/webp';
    44	  if (buf.toString('ascii', 0, 4) === 'GIF8') return 'image/gif';
    45	  const ftyp = buf.toString('ascii', 4, 12);
    46	  if (ftyp.startsWith('ftyp')) {
    47	    const brand = ftyp.slice(4);
    48	    if (['heic', 'heix', 'mif1', 'msf1', 'hevc'].includes(brand)) return 'image/heic';
    49	    if (['avif', 'avis'].includes(brand)) return 'image/avif';
    50	  }
    51	  const head = buf.toString('utf8', 0, Math.min(buf.length, 512)).trimStart();
    52	  if (head.startsWith('<svg') || (head.startsWith('<?xml') && head.includes('<svg'))) return 'image/svg+xml';
    53	  return null;
    54	}

    92	export async function saveUpload(input: Buffer | Uint8Array | ArrayBuffer, opts: SaveUploadOptions): Promise<UploadRecord> {
    93	  let buf: Buffer = Buffer.isBuffer(input) ? input : Buffer.from(input instanceof ArrayBuffer ? new Uint8Array(input) : input);
    94	  if (buf.length > MAX_UPLOAD_BYTES) throw new Error(`이미지가 너무 큽니다 (최대 ${MAX_UPLOAD_BYTES / 1024 / 1024}MB).`);
    95	  let mime = sniffImageMime(buf);
    96	  if (!mime) throw new Error('지원하지 않는 이미지 형식입니다 (JPEG/PNG/WebP/HEIC/AVIF/GIF/SVG).');
    97	  let width: number | null = null;
    98	  let height: number | null = null;
    99	
   100	  if (mime !== 'image/svg+xml') {
   101	    const normalize = opts.normalize ?? true;
   102	    const img = sharp(buf, { failOn: 'none', limitInputPixels: MAX_INPUT_PIXELS });
   103	    if (normalize) {
   104	      const maxEdge = opts.maxEdge ?? MAX_STORED_EDGE;
   105	      const outPng = mime === 'image/png' || mime === 'image/gif';
   106	      const pipeline = img.rotate().resize({ width: maxEdge, height: maxEdge, fit: 'inside', withoutEnlargement: true }).toColorspace('srgb');
   107	      const { data, info } = await (outPng ? pipeline.png() : pipeline.jpeg({ quality: 92, chromaSubsampling: '4:4:4' })).toBuffer({ resolveWithObject: true });
   108	      buf = data;
   109	      mime = outPng ? 'image/png' : 'image/jpeg';
   110	      width = info.width;
   111	      height = info.height;
   112	    } else {
   113	      const meta = await img.metadata();
   114	      width = meta.width ?? null;
   115	      height = meta.height ?? null;
   116	    }
   117	  }
   118	
   119	  const sha256 = crypto.createHash('sha256').update(buf).digest('hex');
   120	  const db = getDb();
   121	  const existing = db.prepare('SELECT * FROM uploads WHERE sha256 = ? AND kind = ? LIMIT 1').get(sha256, opts.kind) as UploadRow | undefined;
   122	  if (existing && fs.existsSync(path.join(uploadsDir(), existing.file_name))) return toRecord(existing);
   123	
   124	  const id = newId('up');
   125	  const fileName = `${id}.${EXT[mime] ?? 'bin'}`;
   126	  fs.writeFileSync(path.join(uploadsDir(), fileName), buf);
   127	  const row: UploadRow = {
   128	    id,
   129	    created_at: nowIso(),
   130	    kind: opts.kind,
   131	    mime,
   132	    bytes: buf.length,
   133	    width,
   134	    height,
   135	    sha256,
   136	    source_url: opts.sourceUrl ?? null,
   137	    file_name: fileName,
   138	  };
   139	  db.prepare(
   140	    'INSERT INTO uploads (id, created_at, kind, mime, bytes, width, height, sha256, source_url, file_name) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)',
   141	  ).run(row.id, row.created_at, row.kind, row.mime, row.bytes, row.width, row.height, row.sha256, row.source_url, row.file_name);
   142	  return toRecord(row);
   143	}

   221	class BodyTooLargeError extends Error {}
   222	
   223	/** 본문을 상한까지만 읽는다. Content-Length가 크면 읽기 전에, 스트림이 상한을 넘으면 그 즉시 중단 */
   224	export async function readBodyCapped(req: Request, max = MAX_UPLOAD_BODY_BYTES): Promise<Uint8Array> {
   225	  const declared = Number(req.headers.get('content-length') ?? '');
   226	  if (Number.isFinite(declared) && declared > max) throw new BodyTooLargeError();
   227	  if (!req.body) return new Uint8Array(0);
   228	  const reader = req.body.getReader();
   229	  const chunks: Uint8Array[] = [];
   230	  let total = 0;
   231	  for (;;) {
   232	    const { done, value } = await reader.read();
   233	    if (done) break;
   234	    total += value.byteLength;
   235	    if (total > max) {
   236	      await reader.cancel().catch(() => {});
   237	      throw new BodyTooLargeError();
   238	    }
   239	    chunks.push(value);
   240	  }
   241	  const out = new Uint8Array(total);
   242	  let o = 0;
   243	  for (const c of chunks) {
   244	    out.set(c, o);
   245	    o += c.byteLength;
   246	  }
   247	  return out;
   248	}

   250	export async function handleUploadPost(req: Request, defaultKind: UploadKind = 'other'): Promise<Response> {
   251	  try {
   252	    const ct = req.headers.get('content-type') ?? '';
   253	    let bytes: Uint8Array;
   254	    try {
   255	      bytes = await readBodyCapped(req);
   256	    } catch (e) {
   257	      if (e instanceof BodyTooLargeError) return jsonError(413, 'too_large', '이미지가 너무 큽니다 (최대 25MB).');
   258	      throw e;
   259	    }
   260	    if (ct.includes('application/json')) {
   261	      const body = JSON.parse(new TextDecoder().decode(bytes)) as { dataUrl?: string; kind?: string };
   262	      if (!body.dataUrl) return jsonError(400, 'bad_request', 'dataUrl이 필요합니다.');
   263	      const kind = UploadKindSchema.catch(defaultKind).parse(body.kind ?? defaultKind);
   264	      return Response.json(await saveUploadFromDataUrl(body.dataUrl, { kind }));
   265	    }
   266	    const form = await new Response(bytes as BodyInit, { headers: { 'content-type': ct } }).formData();
   267	    const file = form.get('file');
   268	    if (!(file instanceof Blob)) return jsonError(400, 'bad_request', "'file' 필드가 필요합니다.");
   269	    if (file.size > MAX_UPLOAD_BYTES) return jsonError(413, 'too_large', '이미지가 너무 큽니다.');
   270	    const kind = UploadKindSchema.catch(defaultKind).parse(String(form.get('kind') ?? defaultKind));
   271	    const rec = await saveUpload(Buffer.from(await file.arrayBuffer()), { kind });
   272	    return Response.json(rec);
   273	  } catch (e) {
   274	    return jsonError(400, 'upload_failed', (e as Error).message);
   275	  }
   276	}

   195	  if (!b) return undefined;

   147	  opts: { stride?: number; minLuma?: number; maxLuma?: number; maxGain?: number } = {},
   148	): WbGains {

   150	  let sr = 0;
   151	  let sg = 0;
   152	  let sb = 0;
   153	  let n = 0;

   155	    const y = 0.299 * r + 0.587 * g + 0.114 * b;
   156	    if (y < minLuma || y > maxLuma) return;

   164	  } else {
   165	    const arr = pixels as ArrayLike<number>;
   166	    const stride = (opts.stride ?? 1) * 4;
   167	    for (let i = 0; i + 2 < arr.length; i += stride) add(arr[i], arr[i + 1], arr[i + 2]);
   168	  }

  - 퍼스널컬러는 측정이 있으면 측정 분류가 기준이다. AI 톤은 교차 검증에만 쓰고 결과로 신뢰도를 조정한다.
  - 측정이 없으면 AI 톤을 쓴다. AI가 읽은 피부·헤어·눈 hex로 축을 계산하는데, 이 축은 celeb-vibe-report의 갭 비교용이다.
  - AI가 톤·피부색을 비우면 규칙 초안 값(사진 중앙 픽셀)으로 되살리지 않는다. '인물 없음·사진 아님'이 확정되면 측정하지 않은 톤과 피부색은 모두 비운다.
  - AI가 아이템·팔레트·룩 이름·분위기를 비웠을 때 규칙 초안으로 채우는 것은 코디(분위기) 분석이 가능한 경우뿐이다. 채운 아이템에는 '규칙 추정' 노트를 단다.
- **결과 표시:** 마지막에 `VibeAnalysisSchema.parse`로 검증한다. `source` 값은 아래와 같다.

### 5. 저장

- **분석 결과:** `vibe_analyses` 테이블에 저장한다. 컬럼은 celeb_id, celeb_name, image_upload_id, analyzable, look_name, is_mock이다.
- **브라우저 측정값:** `kv` 테이블의 `vibe:measurement:<id>`에 저장한다. `다시 분석`은 이 값을 재사용하고 같은 id를 유지한다. 분석 중에 기록이 삭제되면 404로 끝내고 새로 만들지 않는다.
- **사용자 수정값:** 사용자가 고친 룩 이름은 `kv`의 `vibe:edit:<id>`에 따로 두어 `다시 분석` 뒤에도 유지한다. 삭제·초기화 때 함께 지운다.
- **search-celeb 사진:** `celebImageId`로 분석하면 search-celeb의 `celeb_images` 행에 `analyzable`과 사유를 기록한다. 리포트에서 같은 인물의 다른 사진을 고르라고 안내할 때 쓴다.
- **초기화:** 확인 단계에서 지워질 분석 수와, 추구미 가이드가 참조 중인 분석 수를 먼저 보여준다. 업로드 파일은 다른 앱이 참조할 수 있어 지우지 않는다.
