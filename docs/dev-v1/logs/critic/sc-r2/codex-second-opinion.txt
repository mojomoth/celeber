    53	 * - 한쪽의 영문명(nameEn)이 다른 쪽 이름과 같아도 같은 이름으로 본다
    54	 *   (예: 'Jennie'로 검색하면 입력 이름 'Jennie'와 AI 후보 '제니'(nameEn Jennie)를 하나로)
    55	 * - 같은 인물이 여러 출처에서 나오면 점수에 출처당 +0.1 (최대 1)
    56	 * - 근거(evidence)는 합치고 중복 제거
    57	 * - 설명/영문명/썸네일/링크는 먼저 나온 값 유지

    60	export function mergeCandidates(lists: CelebCandidate[][], limit = 8): CelebCandidate[] {
    61	  const entries: { c: CelebCandidate; sources: Set<string>; order: number }[] = [];
    62	  for (const list of lists) {
    63	    for (const c of list) {
    64	      if (!nameKey(c.name)) continue;
    65	      const hit = entries.find((e) => sameCandidate(e.c, c));
    66	      if (!hit) {
    67	        entries.push({ c: { ...c, evidence: [...c.evidence] }, sources: new Set([c.source]), order: entries.length });
    68	        continue;
    69	      }
    70	      const m = hit.c;
    71	      hit.sources.add(c.source);
    72	      m.nameEn ??= c.nameEn;
    73	      m.description ??= c.description;
    74	      m.thumbnailUrl ??= c.thumbnailUrl;
    75	      m.externalUrl ??= c.externalUrl;
    76	      m.score = Math.max(m.score ?? 0, c.score ?? 0);
    77	      for (const e of c.evidence) if (!m.evidence.includes(e)) m.evidence.push(e);
    78	      if (SOURCE_RANK[c.source] < SOURCE_RANK[m.source]) m.source = c.source;
    79	    }
    80	  }
    81	  return entries
    82	    .map(({ c, sources, order: o }) => ({ c: { ...c, score: round2(Math.min(1, (c.score ?? 0.5) + 0.1 * (sources.size - 1))) }, o }))
    83	    .sort((a, b) => (b.c.score ?? 0) - (a.c.score ?? 0) || a.o - b.o)
    84	    .slice(0, limit)
    85	    .map((x) => x.c);
    86	}
