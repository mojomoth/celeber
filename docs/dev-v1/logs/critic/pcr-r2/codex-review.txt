   285	/**
   286	 * AI 서술 표기 정리.
   287	 * - 색온도: '3290 K', '3,290 K' → '3,290K' (리포트의 다른 곳과 같은 표기)
   288	 * - 타입 이름: '보조 타입은 겨울쿨(7%)'처럼 계절만 쓴 경우 규칙 결과의 전체 이름('겨울쿨 트루')으로 고정

   290	export function polishAiText(text: string, ctx?: { tone: Tone; secondaryTone: Tone }): string {
   291	  let t = text
   292	    .replace(/(?<![\d.,])(\d{4,5})\s?K(?![A-Za-z])/g, (_, n: string) => `${Number(n).toLocaleString('en-US')}K`)
   293	    .replace(/(\d,\d{3})\s+K(?![A-Za-z])/g, '$1K');
   294	  if (!ctx) return t;
   295	  const primary = getToneProfile(ctx.tone).nameKo;
   296	  const secondary = getToneProfile(ctx.secondaryTone).nameKo;
   297	  const seasonOf = (name: string) => name.split(' ')[0];
   298	  // '보조 타입은 겨울쿨' → 보조 타입 이름
   299	  t = t.replace(new RegExp(`(보조\\s*(?:가능\\s*)?타입[은는이가]?\\s*)${SEASON_WORD.source}`, 'g'), (m, lead: string, season: string) =>
   300	    seasonOf(secondary) === season ? `${lead}${secondary}` : m,
   301	  );
   302	  // '겨울쿨(7%)'처럼 점수와 함께 쓴 계절 이름 → 1·2순위 중 계절이 하나만 맞으면 그 이름
   303	  t = t.replace(new RegExp(`${SEASON_WORD.source}(?=\\s*\\(\\d)`, 'g'), (m, season: string) => {
   304	    const hits = [...new Set([primary, secondary].filter((n) => seasonOf(n) === season))];
   305	    return hits.length === 1 ? hits[0] : m;
   306	  });
   307	  return t;
   308	}

    95	/** 정면 사진 노출 오류 (camera-face quality 코드) */
    96	const EXPOSURE_ERRORS: Record<string, string> = {
    97	  'face-dark': '정면 얼굴이 너무 어두워',
    98	  'face-bright': '정면 얼굴이 하얗게 날아가',
    99	};
   100	/** 재촬영 권장일 때 웜/쿨 %와 타입 명확도에 거는 신뢰도 상한 */
   101	export const UNRELIABLE_TRUST = 0.4;
   102	
   103	/** 보정 피부값이 사람 피부 범위를 벗어났는지 */
   104	export function skinOutOfRange(face: FaceAnalysis): boolean {
   105	  const s = face.skin.corrected;
   106	  const { lMin, lMax, itaMin, itaMax } = SKIN_PLAUSIBLE;
   107	  return s.lab.l < lMin || s.lab.l > lMax || s.ita < itaMin || s.ita > itaMax;
   108	}
   109	
   110	/**
   111	 * 측정 조건이 판정을 믿기 어려운 수준인지. 다음 중 하나면 재촬영 권장(점수 상한 49, ② 아래 안내, ⑬ 맨 앞).
   112	 * 1) 색이 강한 조명(백열·전구색 LED·형광·혼합, 3,800K 미만/8,000K 초과, 조명 품질 0.45 미만)에서 피부 부위 간 편차까지 큼
   113	 * 2) 정면 사진에 노출 오류('얼굴이 너무 어둡습니다' 등)가 있는데 그대로 진행함
   114	 * 3) 보정 피부 L*나 ITA가 사람 피부 범위를 벗어남
   115	 * 분류기 1·2위 차가 커도(확신이 높아 보여도) 신뢰도가 '보통' 이상으로 나오지 않게 한다.
   116	 */
   117	export function unreliableCapture(face: FaceAnalysis): { unreliable: boolean; reason: string } {
   118	  const env = face.environment;
   119	  const harsh = HARSH_LIGHT.has(env.lightSource) || env.cctK < 3800 || env.cctK > 8000 || env.lightingQuality < 0.45;
   120	  const spread = face.skin.regionSpreadDeltaE;
   121	  const causes: string[] = [];
   122	  const exposure = face.front.quality.issues.find((i) => i.severity === 'error' && EXPOSURE_ERRORS[i.code]);
   123	  if (exposure) causes.push(`${EXPOSURE_ERRORS[exposure.code]} 피부색을 제대로 잴 수 없었습니다.`);
   124	  if (skinOutOfRange(face)) {
   125	    const s = face.skin.corrected;
   126	    causes.push(`측정된 피부값(L* ${s.lab.l.toFixed(1)}, ITA ${Math.round(s.ita)}°)이 사람 피부에서 나오기 어려운 값입니다.`);
   127	  }
   128	  if (harsh && spread > SKIN_SPREAD_UNRELIABLE_DE) {
   129	    causes.push(`${env.lightSourceLabel}(${env.cctK.toLocaleString('en-US')}K) 아래에서 이마·볼·턱 색 편차(ΔE ${spread.toFixed(1)})가 커서 웜/쿨 판정이 흔들릴 수 있습니다.`);
   130	  }
   131	  if (!causes.length) return { unreliable: false, reason: '' };
   132	  const fix = exposure || causes.length > 1 || skinOutOfRange(face) ? '밝은 자연광을 마주 보고 정면에서 다시 촬영해 주세요.' : '자연광 정면에서 다시 촬영해 주세요.';
   133	  return { unreliable: true, reason: `${causes.join(' ')} ${fix}` };
   134	}
   135	
   136	/**
   137	 * 조명 자체가 판정을 흔들 수 있는 촬영인지 (부위 편차와 무관하게).
   138	 * 환경 단계가 '재촬영 권장'을 냈거나, 조명 품질이 0.45 미만이거나, 색이 강한 인공 조명이면 true.
   139	 * 이때 ③ 웜/쿨 %와 ⑭ 타입 구분 명확도를 조명 신뢰도(trust)만큼 50:50 쪽으로 줄이고, 점수 상한을 둔다.
   140	 */
   141	export function lightingCaution(face: FaceAnalysis): { flagged: boolean; trust: number; reason: string } {
   142	  const env = face.environment;
   143	  const harsh = HARSH_LIGHT.has(env.lightSource) || env.cctK < 3800 || env.cctK > 8000;
   144	  const flagged = env.retakeRecommended || env.lightingQuality < 0.45 || harsh;
   145	  if (!flagged) return { flagged: false, trust: 1, reason: '' };
   146	  const lq = Math.max(0, Math.min(1, env.lightingQuality));
   147	  const trust = Math.round((0.4 + 0.4 * Math.min(1, lq / 0.75)) * 100) / 100;
   148	  const why = [
   149	    env.retakeRecommended ? '환경 촬영에서 재촬영 권장' : '',
   150	    `${env.lightSourceLabel} ${env.cctK.toLocaleString('en-US')}K`,
   151	    `조명 품질 ${Math.round(lq * 100)}점`,
   152	  ]
   153	    .filter(Boolean)
   154	    .join(', ');
   155	  return {
   156	    flagged,
   157	    trust,
   158	    reason: `조명 확인 필요: ${why}. ${harsh ? '색이 있는 조명에서는' : '노출·조명 상태가 나쁘면'} 보정 후에도 웜/쿨이 바뀔 수 있어 웜/쿨 비율과 타입 확신도를 낮춰 표시했습니다. 자연광 정면에서 다시 촬영해 보세요.`,
   159	  };
   160	}
   161	
   162	/** 웜/쿨 %와 타입 명확도에 쓰는 측정 신뢰도: 조명 경고면 조명 신뢰도, 재촬영 권장이면 최대 0.4 */
   163	export function captureTrust(face: FaceAnalysis): number {
   164	  const t = lightingCaution(face).trust;
   165	  return unreliableCapture(face).unreliable ? Math.min(t, UNRELIABLE_TRUST) : t;
   166	}

   168	/** 조명 신뢰도만큼 웜 %를 50 쪽으로 당긴다 (합 100 유지) */
   169	export function softenWarmPercent(warm: number, trust: number): { warm: number; cool: number } {
   170	  const w = Math.round(50 + (warm - 50) * Math.max(0, Math.min(1, trust)));
   171	  return { warm: w, cool: 100 - w };
   172	}

   174	/** 조명 경고 시 점수 상한 */
   175	export const FLAGGED_LIGHTING_SCORE_CAP = 55;
   176	
   177	/** 사진 밝기가 적정이 아니면 조명 요인 점수에 거는 상한 */
   178	const BRIGHTNESS_LIGHT_CAP: Record<string, number> = { 'too-dark': 0.45, dim: 0.7, bright: 0.7, 'too-bright': 0.45 };
   179	const BRIGHTNESS_TEXT: Record<string, string> = {
   180	  'too-dark': '사진이 너무 어둡습니다',
   181	  dim: '사진이 조금 어둡습니다',
   182	  bright: '사진이 조금 밝습니다',
   183	  'too-bright': '사진이 너무 밝아 하얗게 날아간 부분이 있습니다',
   184	};
   185	
   186	/** ⑭ '조명 조건' 근거: 조명 품질과 사진 밝기를 함께 본다 ('조명이 고르고 밝기가 적정합니다 (… 조금 어두움)' 같은 모순 방지) */
   187	export function lightingDetail(env: FaceAnalysis['environment'], lq: number): string {
   188	  const where = `${env.lightSourceLabel} · ${env.cctK.toLocaleString('en-US')}K`;
   189	  const bright = BRIGHTNESS_TEXT[env.brightness.level];
   190	  if (!bright) return `${lq >= 0.75 ? '조명이 고르고 밝기가 적정합니다.' : lq >= 0.5 ? '조명에 일부 제약이 있습니다.' : '조명 조건이 좋지 않습니다.'} (${where})`;
   191	  const head = lq >= 0.75 ? '조명은 고른 편이지만' : lq >= 0.5 ? '조명에 일부 제약이 있고' : '조명 조건이 좋지 않고';
   192	  return `${head} ${bright}. (${where})`;
   193	}

   195	/**
   196	 * ⑭ 피부 측정 품질 근거: camera-face의 내부 사유(게인·강도·픽셀 수)를 사용자 문장으로 바꾼다.
   197	 * 문제를 알리는 사유를 앞에 두고 최대 2개만 쓴다.
   198	 */
   199	export function skinReasonTexts(reasons: readonly string[]): string[] {
   200	  const problems: string[] = [];
   201	  const notes: string[] = [];
   202	  for (const r of reasons) {
   203	    let m: RegExpExecArray | null;
   204	    if (/조명 추정 신뢰도가 낮아/.test(r)) problems.push('조명 추정이 불확실해 색 보정을 약하게 적용했습니다.');
   205	    else if ((m = /^촬영 품질 오류\((.+?)\)/.exec(r))) problems.push(`촬영 문제(${m[1]})가 있어 점수를 낮췄습니다.`);
   206	    else if (/^촬영 품질 경고/.test(r)) problems.push('촬영 품질 경고가 있어 점수를 낮췄습니다.');
   207	    else if (/^피부 샘플 픽셀이 적습니다/.test(r)) problems.push('피부가 보이는 면적이 작습니다. 얼굴을 더 크게 찍으면 정확해집니다.');
   208	    else if (/^일부 피부 영역/.test(r)) problems.push('이마·볼·턱 중 일부를 측정하지 못했습니다.');
   209	    else if (/보정량이 (커서|큽니다)/.test(r)) problems.push('조명 색 보정량이 커서 원래 조명의 영향이 남아 있을 수 있습니다.');
   210	    else if (/^피부 밝기 편차/.test(r)) problems.push('그림자나 반사광 때문에 피부 밝기 편차가 큽니다.');
   211	    else if (/^보정에 쓸 배경 정보가 없어/.test(r)) problems.push('조명 색을 추정할 배경이 없어 원본 색을 그대로 썼습니다.');
   212	    else if (/^흰 종이 기준 백색/.test(r)) notes.push('흰 종이를 기준으로 색을 보정했습니다.');
   213	    else if (/gray-world 게인 기하 평균/.test(r)) notes.push('환경 사진과 얼굴 배경으로 조명 색을 추정해 보정했습니다.');
   214	    else if ((m = /^(환경 사진|얼굴 사진 배경) gray-world/.exec(r))) notes.push(`${m[1] === '환경 사진' ? '환경 사진' : '얼굴 배경'}으로 조명 색을 추정해 보정했습니다.`);
   215	    // 그 밖(픽셀 수, 조명 품질 반영, 마스크 사용, 부위 편차)은 다른 요인과 겹치거나 내부 정보라 뺀다.
   216	  }
   217	  return [...problems, ...notes].slice(0, 2);
   218	}

   220	/** 정면 촬영 오류 코드 → 리포트 근거용 짧은 이름 (camera-face quality 코드) */
   221	const CAPTURE_ISSUE_LABEL: Record<string, string> = {
   222	  'face-dark': '얼굴 어두움',
   223	  'face-bright': '얼굴 과다 노출',
   224	  'face-cut': '얼굴 잘림',
   225	  'too-small': '얼굴이 작음',
   226	  'too-large': '얼굴이 너무 가까움',
   227	  yaw: '고개 좌우 각도',
   228	  pitch: '고개 위아래 각도',
   229	  roll: '고개 기울어짐',
   230	  'off-center': '얼굴 위치',
   231	  'no-face': '얼굴 미검출',
   232	};

   234	/** ⑭ 정면 검출·촬영 품질: 촬영 오류 1건마다 0.7배 (강행 촬영이 점수에 드러나게) */
   235	export function captureQualityScore(q: FaceAnalysis['front']['quality']): number {
   236	  const errors = q.issues.filter((i) => i.severity === 'error').length;
   237	  return q.detectionScore * (q.passed ? 1 : 0.8) * 0.7 ** errors;
   238	}

   240	/** ⑭ 실제 측정 품질에서 신뢰도 계산 (고정값 없음) */
   241	export function reportConfidence(face: FaceAnalysis, cls: PersonalColorClassification, aiTone: Tone | null | undefined): ReportConfidence {
   242	  const q = face.front.quality;
   243	  const env = face.environment;
   244	  const light = lightingCaution(face);
   245	  const u = unreliableCapture(face);
   246	  const trust = captureTrust(face);
   247	  const lq = Math.min(env.lightingQuality, BRIGHTNESS_LIGHT_CAP[env.brightness.level] ?? 1);
   248	  // 신뢰도가 낮으면 1·2위 차를 그대로 믿지 않는다. 0.35에서 만점이므로 먼저 자르고 곱해야 표시 점수에도 감점이 보인다.
   249	  const margin = Math.min(cls.margin, 0.35) * trust;
   250	  const base = computeReportConfidence({
   251	    lightingQuality: lq,
   252	    whiteBalanceDeltaE: face.skin.whiteBalance.deltaE,
   253	    whiteBalanceMethod: face.skin.whiteBalance.method,
   254	    skinConfidence: face.skin.confidence,
   255	    faceDetectionScore: captureQualityScore(q),
   256	    skinSpreadDeltaE: face.skin.regionSpreadDeltaE,
   257	    classifierMargin: margin,
   258	    ruleAiAgreement: ruleAiAgreement(cls.tone, aiTone),
   259	    hasHair: !!face.featureColors.hair,
   260	    hasEyes: !!face.featureColors.eyes,
   261	  });
   262	  // 근거 문구에 실제 측정값을 덧붙인다
   263	  const errorLabels = [...new Set(q.issues.filter((i) => i.severity === 'error').map((i) => CAPTURE_ISSUE_LABEL[i.code] ?? '촬영 기준 미달'))];
   264	  const factors: ConfidenceFactor[] = base.factors.map((f) => {
   265	    if (f.key === 'lighting') return { ...f, detail: lightingDetail(env, lq) };
   266	    if (f.key === 'face-detection') {
   267	      const issues = errorLabels.length ? `, ${errorLabels.join(' · ')}` : '';
   268	      return { ...f, detail: `${f.detail} (검출 ${Math.round(q.detectionScore * 100)}%, 좌우 ${Math.round(face.front.yawDeg)}°${issues})` };
   269	    }
   270	    if (f.key === 'classifier-margin') {
   271	      const gap = Math.round(cls.margin * 100);
   272	      if (trust >= 1) return { ...f, detail: `${f.detail} (1·2위 차 ${gap}%p)` };
   273	      const why = u.unreliable ? '촬영 상태를 고려해' : '조명 영향을 고려해';
   274	      if (cls.margin / 0.35 >= 0.57) return { ...f, detail: `1·2위 차는 ${gap}%p지만 ${why} 확신도를 낮춰 반영했습니다.` };
   275	      return { ...f, detail: `보조 타입과 점수가 가깝습니다(1·2위 차 ${gap}%p). ${why} 확신도를 더 낮췄습니다.` };
   276	    }
   277	    if (f.key === 'skin-uniformity' && u.unreliable && face.skin.regionSpreadDeltaE > SKIN_SPREAD_UNRELIABLE_DE) return { ...f, detail: `${f.detail} (조명 얼룩 의심)` };
   278	    if (f.key === 'skin-measurement') {
   279	      const extra = skinReasonTexts(face.skin.confidenceReasons);
   280	      return extra.length ? { ...f, detail: `${f.detail} ${extra.join(' ')}` } : f;
   281	    }
   282	    return f;
   283	  });
   284	  if (u.unreliable) {
   285	    // 앱 단 감점: 점수의 70%, 최대 49 → 항상 '낮은 신뢰도'. 점수는 여전히 측정값에 따라 달라진다.
   286	    const score = Math.min(49, Math.round(base.score * 0.7));
   287	    return { score, grade: 'low', gradeLabel: '낮은 신뢰도 · 재촬영 권장', factors };
   288	  }
   289	  if (light.flagged) {
   290	    // 상한 55로 부드럽게 눌러(50 초과분은 10%만 반영) 측정 품질에 따른 순서는 유지한다.
   291	    const score = base.score <= 50 ? base.score : Math.min(FLAGGED_LIGHTING_SCORE_CAP, Math.round(50 + (base.score - 50) * 0.1));
   292	    const grade = score >= 50 ? 'medium' : 'low';
   293	    return { score, grade, gradeLabel: grade === 'medium' ? '보통 신뢰도 · 조명 확인 필요' : '낮은 신뢰도 · 조명 확인 필요', factors };
   294	  }
   295	  return { ...base, factors };
   296	}

   339	  const light = lightingCaution(face);
