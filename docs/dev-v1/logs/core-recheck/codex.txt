    95	function collect(
    96	  img: ImageDataLike,
    97	  include: Pt[],
    98	  exclude: Pt[][],
    99	  filter: ((r: number, g: number, b: number, x: number, y: number) => boolean) | null,
   100	  maxPixels: number,
   101	): Rgb[] {
   102	  if (include.length < 3) return [];
   103	  const bb = boundingBox(include);
   104	  const x0 = Math.max(0, Math.floor(bb.x));
   105	  const y0 = Math.max(0, Math.floor(bb.y));
   106	  const x1 = Math.min(img.width - 1, Math.ceil(bb.x + bb.w));
   107	  const y1 = Math.min(img.height - 1, Math.ceil(bb.y + bb.h));
   108	  const area = Math.max(1, (x1 - x0 + 1) * (y1 - y0 + 1));
   109	  // ceil: 격자 간격을 올림해야 상한을 넘지 않는다 (floor면 면적이 상한의 4배 미만일 때 step 1이 된다)
   110	  const step = Math.max(1, Math.ceil(Math.sqrt(area / maxPixels)));
   111	  const exBoxes = exclude.map((p) => ({ poly: p, bb: boundingBox(p) }));
   112	  const out: Rgb[] = [];
   113	  for (let y = y0; y <= y1; y += step) {
   114	    for (let x = x0; x <= x1; x += step) {
   115	      const p = { x: x + 0.5, y: y + 0.5 };
   116	      if (!pointInPolygon(p, include)) continue;
   117	      let skip = false;
   118	      for (const e of exBoxes) {
   119	        if (p.x < e.bb.x || p.x > e.bb.x + e.bb.w || p.y < e.bb.y || p.y > e.bb.y + e.bb.h) continue;
   120	        if (pointInPolygon(p, e.poly)) {
   121	          skip = true;
   122	          break;
   123	        }
   124	      }
   125	      if (skip) continue;
   126	      const i = (y * img.width + x) * 4;
   127	      const r = img.data[i];
   128	      const g = img.data[i + 1];
   129	      const b = img.data[i + 2];

   131	      out.push({ r, g, b });
   132	    }
   133	  }

   306	  }
   307

   310	    corrected,
   311	    regions,
   312	    sampleCount: samples.skin.overall.used,
   313	    trim,
   314	    whiteBalance: {
   315	      method: wb.method,
   316	      gains: { r: wb.gains.r, g: wb.gains.g, b: wb.gains.b },
   317	      strength: wb.strength,
   318	      sourceCctK: wb.sourceCctK,
   319	      deltaE: Math.round(wbDelta * 10) / 10,
   320	    },
   321	    regionSpreadDeltaE: Math.round(regionSpreadDeltaE * 10) / 10,
   322	    confidence: Math.max(0.05, Math.min(1, conf)),
   323	    confidenceReasons: reasons,
   324	  };

    73	/** 매직 바이트로 이미지 MIME 판별 (확장자/헤더 신뢰 안 함) */
    74	export function sniffImageMime(buf: Buffer): string | null {
    75	  if (buf.length < 12) return null;
    76	  if (buf[0] === 0xff && buf[1] === 0xd8 && buf[2] === 0xff) return 'image/jpeg';
    77	  if (buf[0] === 0x89 && buf[1] === 0x50 && buf[2] === 0x4e && buf[3] === 0x47) return 'image/png';
    78	  if (buf.toString('ascii', 0, 4) === 'RIFF' && buf.toString('ascii', 8, 12) === 'WEBP') return 'image/webp';
    79	  if (buf.toString('ascii', 0, 4) === 'GIF8') return 'image/gif';
    80	  const ftyp = buf.toString('ascii', 4, 12);
    81	  if (ftyp.startsWith('ftyp')) {
    82	    const brand = ftyp.slice(4);
    83	    if (['heic', 'heix', 'mif1', 'msf1', 'hevc'].includes(brand)) return 'image/heic';
    84	    if (['avif', 'avis'].includes(brand)) return 'image/avif';
    85	  }
    86	  const head = buf.toString('utf8', 0, Math.min(buf.length, 512)).trimStart();
    87	  if (head.startsWith('<svg') || (head.startsWith('<?xml') && head.includes('<svg'))) return 'image/svg+xml';
    88	  return null;
    89	}

   126	/** 이미지 버퍼 저장 → UploadRecord. 이미지가 아니거나 디코드할 수 없으면 UploadError. */
   127	export async function saveUpload(input: Buffer | Uint8Array | ArrayBuffer, opts: SaveUploadOptions): Promise<UploadRecord> {
   128	  let buf: Buffer = Buffer.isBuffer(input) ? input : Buffer.from(input instanceof ArrayBuffer ? new Uint8Array(input) : input);
   129	  if (buf.length > MAX_UPLOAD_BYTES) throw new UploadError(413, 'too_large', UPLOAD_MESSAGES.tooLarge);
   130	  let mime = sniffImageMime(buf);
   131	  if (!mime) throw new UploadError(400, 'unsupported_type', UPLOAD_MESSAGES.unsupported);
   132	  let width: number | null = null;
   133	  let height: number | null = null;
   134	
   135	  if (mime === 'image/svg+xml') {
   136	    if (RASTER_ONLY_KINDS.has(opts.kind)) throw new UploadError(422, 'svg_not_allowed', UPLOAD_MESSAGES.svgNotAllowed);
   137	    // 원본을 그대로 저장하지만, 깨진 XML이 저장된 뒤 분석 단계에서 500이 되지 않도록 여기서 파싱해 본다.
   138	    // 크기 선언이 큰 SVG(목업 등)는 렌더하지 않으므로 픽셀 상한은 걸지 않는다.
   139	    try {
   140	      await sharp(buf, { limitInputPixels: false }).metadata();
   141	    } catch (e) {
   142	      throw decodeError(e, mime);
   143	    }
   144	  } else {
   145	    try {
   146	      ({ buf, mime, width, height } = await normalizeRaster(buf, mime, opts));
   147	    } catch (e) {
   148	      throw decodeError(e, mime);
   149	    }
   150	  }
   151	  return storeUpload(buf, mime, width, height, opts);
   152	}

   154	/** 래스터: EXIF 회전 + sRGB + 최대 변 제한 (normalize=false면 크기만 읽는다) */
   155	async function normalizeRaster(buf: Buffer, mime: string, opts: SaveUploadOptions): Promise<{ buf: Buffer; mime: string; width: number | null; height: number | null }> {
   156	  const img = sharp(buf, { failOn: 'none', limitInputPixels: MAX_INPUT_PIXELS });
   157	  if (!(opts.normalize ?? true)) {
   158	    const meta = await img.metadata();
   159	    return { buf, mime, width: meta.width ?? null, height: meta.height ?? null };
   160	  }
   161	  const maxEdge = opts.maxEdge ?? MAX_STORED_EDGE;
   162	  const outPng = mime === 'image/png' || mime === 'image/gif';
   163	  const pipeline = img.rotate().resize({ width: maxEdge, height: maxEdge, fit: 'inside', withoutEnlargement: true }).toColorspace('srgb');
   164	  const { data, info } = await (outPng ? pipeline.png() : pipeline.jpeg({ quality: 92, chromaSubsampling: '4:4:4' })).toBuffer({ resolveWithObject: true });
   165	  return { buf: data, mime: outPng ? 'image/png' : 'image/jpeg', width: info.width, height: info.height };
   166	}

   169	  const sha256 = crypto.createHash('sha256').update(buf).digest('hex');
   170	  const db = getDb();

   173	
   174	  const id = newId('up');

   266	/** 업로드 요청 본문 상한: 25MB 이미지의 base64(×4/3) + 여유 */
   267	export const MAX_UPLOAD_BODY_BYTES = Math.ceil(MAX_UPLOAD_BYTES * 1.4) + 1024 * 1024;
   268	class BodyTooLargeError extends Error {}
   269	
   270	/** 본문을 상한까지만 읽는다. Content-Length가 크면 읽기 전에, 스트림이 상한을 넘으면 그 즉시 중단 */
   271	export async function readBodyCapped(req: Request, max = MAX_UPLOAD_BODY_BYTES): Promise<Uint8Array> {
   272	  const declared = Number(req.headers.get('content-length') ?? '');
   273	  if (Number.isFinite(declared) && declared > max) throw new BodyTooLargeError();
   274	  if (!req.body) return new Uint8Array(0);
   275	  const reader = req.body.getReader();
   276	  const chunks: Uint8Array[] = [];
   277	  let total = 0;
   278	  for (;;) {
   279	    const { done, value } = await reader.read();
   280	    if (done) break;
   281	    total += value.byteLength;
   282	    if (total > max) {
   283	      await reader.cancel().catch(() => {});
   284	      throw new BodyTooLargeError();
   285	    }
   286	    chunks.push(value);
   287	  }
   288	  const out = new Uint8Array(total);
   289	  let o = 0;
   290	  for (const c of chunks) {
   291	    out.set(c, o);
   292	    o += c.byteLength;
   293	  }
   294	  return out;
   295	}

   297	export async function handleUploadPost(req: Request, defaultKind: UploadKind = 'other'): Promise<Response> {
   298	  try {
   299	    const ct = req.headers.get('content-type') ?? '';
   300	    let bytes: Uint8Array;
   301	    try {
   302	      bytes = await readBodyCapped(req);
   303	    } catch (e) {
   304	      if (e instanceof BodyTooLargeError) return jsonError(413, 'too_large', UPLOAD_MESSAGES.tooLarge);
   305	      return jsonError(400, 'upload_failed', UPLOAD_MESSAGES.bodyUnreadable);
   306	    }
   307	    if (ct.includes('application/json')) {
   308	      let body: { dataUrl?: string; kind?: string };
   309	      try {
   310	        body = JSON.parse(new TextDecoder().decode(bytes)) as typeof body;
   311	      } catch {
   312	        return jsonError(400, 'invalid_json', '요청 본문이 JSON이 아닙니다.');
   313	      }
   314	      if (!body?.dataUrl) return jsonError(400, 'bad_request', 'dataUrl이 필요합니다.');
   315	      const kind = UploadKindSchema.catch(defaultKind).parse(body.kind ?? defaultKind);
   316	      return Response.json(await saveUploadFromDataUrl(body.dataUrl, { kind }));
   317	    }
   318	    let form: FormData;
   319	    try {
   320	      form = await new Response(bytes as BodyInit, { headers: { 'content-type': ct } }).formData();
   321	    } catch {
   322	      // 런타임 원문('Failed to parse body as FormData.')을 내보내지 않는다. 대개 본문이 잘린 경우다
   323	      return jsonError(400, 'upload_failed', UPLOAD_MESSAGES.bodyUnreadable);
   324	    }
   325	    const file = form.get('file');
   326	    if (!(file instanceof Blob)) return jsonError(400, 'bad_request', "'file' 필드가 필요합니다.");
   327	    if (file.size > MAX_UPLOAD_BYTES) return jsonError(413, 'too_large', UPLOAD_MESSAGES.tooLarge);
   328	    const kind = UploadKindSchema.catch(defaultKind).parse(String(form.get('kind') ?? defaultKind));
   329	    const rec = await saveUpload(Buffer.from(await file.arrayBuffer()), { kind });
   330	    return Response.json(rec);
   331	  } catch (e) {
   332	    if (e instanceof UploadError) return jsonError(e.status, e.code, e.message);
   333	    console.error('[celeber] upload error', e);
   334	    return jsonError(500, 'upload_failed', UPLOAD_MESSAGES.failed);
   335	  }
   336	}

    95	function collect(
    96	  img: ImageDataLike,
    97	  include: Pt[],
    98	  exclude: Pt[][],
    99	  filter: ((r: number, g: number, b: number, x: number, y: number) => boolean) | null,
   100	  maxPixels: number,
   101	): Rgb[] {
   102	  if (include.length < 3) return [];
   103	  const bb = boundingBox(include);
   104	  const x0 = Math.max(0, Math.floor(bb.x));
   105	  const y0 = Math.max(0, Math.floor(bb.y));
   106	  const x1 = Math.min(img.width - 1, Math.ceil(bb.x + bb.w));
   107	  const y1 = Math.min(img.height - 1, Math.ceil(bb.y + bb.h));
   108	  const area = Math.max(1, (x1 - x0 + 1) * (y1 - y0 + 1));
   109	  // ceil: 격자 간격을 올림해야 상한을 넘지 않는다 (floor면 면적이 상한의 4배 미만일 때 step 1이 된다)
   110	  const step = Math.max(1, Math.ceil(Math.sqrt(area / maxPixels)));
   111	  const exBoxes = exclude.map((p) => ({ poly: p, bb: boundingBox(p) }));
   112	  const out: Rgb[] = [];
   113	  for (let y = y0; y <= y1; y += step) {
   114	    for (let x = x0; x <= x1; x += step) {
   115	      const p = { x: x + 0.5, y: y + 0.5 };
   116	      if (!pointInPolygon(p, include)) continue;
   117	      let skip = false;
   118	      for (const e of exBoxes) {
   119	        if (p.x < e.bb.x || p.x > e.bb.x + e.bb.w || p.y < e.bb.y || p.y > e.bb.y + e.bb.h) continue;
   120	        if (pointInPolygon(p, e.poly)) {
   121	          skip = true;
   122	          break;
   123	        }
   124	      }
   125	      if (skip) continue;
   126	      const i = (y * img.width + x) * 4;
   127	      const r = img.data[i];
   128	      const g = img.data[i + 1];
   129	      const b = img.data[i + 2];
   130	      if (filter && !filter(r, g, b, x, y)) continue;
   131	      out.push({ r, g, b });
   132	    }
   133	  }
   134	  return out;
   135	}

   241	export function reportConfidence(face: FaceAnalysis, cls: PersonalColorClassification, aiTone: Tone | null | undefined): ReportConfidence {
   242	  const q = face.front.quality;
   243	  const env = face.environment;
   244	  const light = lightingCaution(face);
   245	  const u = unreliableCapture(face);
   246	  const trust = captureTrust(face);
   247	  const lq = Math.min(env.lightingQuality, BRIGHTNESS_LIGHT_CAP[env.brightness.level] ?? 1);
   248	  // 신뢰도가 낮으면 1·2위 차를 그대로 믿지 않는다. 0.35에서 만점이므로 먼저 자르고 곱해야 표시 점수에도 감점이 보인다.
   249	  const margin = Math.min(cls.margin, 0.35) * trust;
   250	  const base = computeReportConfidence({
   251	    lightingQuality: lq,
   252	    whiteBalanceDeltaE: face.skin.whiteBalance.deltaE,
   253	    whiteBalanceMethod: face.skin.whiteBalance.method,
   254	    skinConfidence: face.skin.confidence,
   255	    faceDetectionScore: captureQualityScore(q),
   256	    skinSpreadDeltaE: face.skin.regionSpreadDeltaE,
   257	    classifierMargin: margin,
   258	    ruleAiAgreement: ruleAiAgreement(cls.tone, aiTone),
   259	    hasHair: !!face.featureColors.hair,
   260	    hasEyes: !!face.featureColors.eyes,
   261	  });
   262	  // 근거 문구에 실제 측정값을 덧붙인다
   263	  const errorLabels = [...new Set(q.issues.filter((i) => i.severity === 'error').map((i) => CAPTURE_ISSUE_LABEL[i.code] ?? '촬영 기준 미달'))];
   264	  const factors: ConfidenceFactor[] = base.factors.map((f) => {
   265	    if (f.key === 'lighting') return { ...f, detail: lightingDetail(env, lq) };
   266	    if (f.key === 'face-detection') {
   267	      const issues = errorLabels.length ? `, ${errorLabels.join(' · ')}` : '';
   268	      return { ...f, detail: `${f.detail} (검출 ${Math.round(q.detectionScore * 100)}%, 좌우 ${Math.round(face.front.yawDeg)}°${issues})` };
   269	    }
   270	    if (f.key === 'classifier-margin') {
   271	      const gap = Math.round(cls.margin * 100);
   272	      if (trust >= 1) return { ...f, detail: `${f.detail} (1·2위 차 ${gap}%p)` };
   273	      const why = u.unreliable ? '촬영 상태를 고려해' : '조명 영향을 고려해';
   274	      if (cls.margin / 0.35 >= 0.57) return { ...f, detail: `1·2위 차는 ${gap}%p지만 ${why} 확신도를 낮춰 반영했습니다.` };
   275	      return { ...f, detail: `보조 타입과 점수가 가깝습니다(1·2위 차 ${gap}%p). ${why} 확신도를 더 낮췄습니다.` };
   276	    }
   277	    if (f.key === 'skin-uniformity' && u.unreliable && face.skin.regionSpreadDeltaE > SKIN_SPREAD_UNRELIABLE_DE) return { ...f, detail: `${f.detail} (조명 얼룩 의심)` };
   278	    if (f.key === 'skin-measurement') {
   279	      const extra = skinReasonTexts(face.skin.confidenceReasons);
   280	      return extra.length ? { ...f, detail: `${f.detail} ${extra.join(' ')}` } : f;
   281	    }
   282	    return f;
   283	  });
   284	  if (u.unreliable) {
   285	    // 앱 단 감점: 점수의 70%, 최대 49 → 항상 '낮은 신뢰도'. 점수는 여전히 측정값에 따라 달라진다.
   286	    const score = Math.min(49, Math.round(base.score * 0.7));
   287	    return { score, grade: 'low', gradeLabel: '낮은 신뢰도 · 재촬영 권장', factors };
   288	  }
   289	  if (light.flagged) {
   290	    // 상한 55로 부드럽게 눌러(50 초과분은 10%만 반영) 측정 품질에 따른 순서는 유지한다.
   291	    const score = base.score <= 50 ? base.score : Math.min(FLAGGED_LIGHTING_SCORE_CAP, Math.round(50 + (base.score - 50) * 0.1));
   292	    const grade = score >= 50 ? 'medium' : 'low';
   293	    return { score, grade, gradeLabel: grade === 'medium' ? '보통 신뢰도 · 조명 확인 필요' : '낮은 신뢰도 · 조명 확인 필요', factors };
   294	  }
   295	  return { ...base, factors };
   296	}

   103	export function convexHull(points: readonly Pt[]): Pt[] {
   104	  const pts = [...points].sort((a, b) => a.x - b.x || a.y - b.y);
   105	  if (pts.length <= 2) return pts;
   106	  const cross = (o: Pt, a: Pt, b: Pt) => (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);
   107	  const lower: Pt[] = [];

- `9000×5000` SVG를 `kind=vibe`로 올리면 업로드는 `200`입니다. 발급된 ID로 분석하면 픽셀 상한에 걸리고, 이미 존재하는 이미지에 `404 upload_not_found`를 반환합니다. [uploads.ts:140](/Users/jyl/work/repo/celeber/pocs/core/src/server/uploads.ts:140), [celeb-vibe/index.ts:54](/Users/jyl/work/repo/celeber/pocs/celeb-vibe/src/server/index.ts:54)
- 피부색으로 채운 `232×229` 직사각형 ROI에서 기본 상한은 `6,000px`인데 `6,006px`을 수집합니다. [sample.ts:110](/Users/jyl/work/repo/celeber/pocs/core/src/face/sample.ts:110)
- 밝은 이마 `30px`과 어두운 턱 `1,000px`만 남으면, 기본 휘도 컷이 이마 표본을 전부 제거합니다. 균등 가중으로 명시된 전체 피부색이 턱 색 `#8D5A45`로 나옵니다. [sample.ts:176](/Users/jyl/work/repo/celeber/pocs/core/src/face/sample.ts:176)
- 오른쪽 볼만 누락되면 피부 신뢰도가 `0.1` 깎이지만, 리포트는 `일부 피부 영역`으로 시작하는 문구만 인식해 `오른쪽 볼 피부색을 읽지 못해…`라는 감점 근거를 표시하지 않습니다. [sample.ts:285](/Users/jyl/work/repo/celeber/pocs/core/src/face/sample.ts:285), [assemble.ts:208](/Users/jyl/work/repo/celeber/pocs/personal-color-report/src/lib/assemble.ts:208)
